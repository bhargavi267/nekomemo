async function genCards(id) { const t = topicById(id); if (!t) return; const button = document.querySelector(`[data-generate="${id}"]`); if (button) { button.disabled = true; button.textContent = "Generating flashcards..."; } try { const response = await fetch("/.netlify/functions/generate-cards", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ topic: t.title, subject: t.subject, level: t.level, purpose: t.purpose, context: t.lesson?.context || "", count: t.count || 10 }) }); const result = await response.json(); if (!response.ok) { throw new Error(result.error || "Could not generate flashcards."); } if (!Array.isArray(result.cards) || result.cards.length === 0) { throw new Error("No flashcards were returned. Please try again."); } const existing = new Set( data.cards .filter(c => c.topicId === id) .map(c => c.question.trim().toLowerCase()) ); let added = 0; result.cards.forEach(card => { const question = String(card.front || card.question || "").trim(); const answer = String(card.back || card.answer || "").trim(); if ( !question || !answer || existing.has(question.toLowerCase()) ) { return; } existing.add(question.toLowerCase()); data.cards.push({ id: uid(), subject: t.subject, topicId: id, question, answer, reading: "", difficulty: "Normal", notes: "", important: false, source: "AI generated", createdAt: new Date().toISOString(), reviews: 0, nextReview: today() }); added++; }); if (added === 0) { toast("No new cards generated. Try another topic."); return; } save(); toast(`${added} AI flashcards created!`); go("flashcards"); } catch (error) { console.error("Flashcard generation failed:", error); toast(error.message || "Generation failed. Please try again."); } finally { if (button) { button.disabled = false; button.textContent = "Generate starter flashcards →"; } } }
